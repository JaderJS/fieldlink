import { randomUUIDv7 } from "bun";

export function sanitizeFilename(filename: string): string {
	const normalized = filename
		.normalize("NFKD")
		.replace(/\p{Diacritic}/gu, "")
		.replace(/[/\\]/g, "_")
		.replace(/[^\p{L}\p{N}._-]/gu, "_")
		.replace(/\.{2,}/g, ".")
		.replace(/^[._-]+|[._-]+$/g, "")
		.slice(0, 180);

	return normalized || "file";
}

export function createObjectKey(filename: string, prefix = "uploads"): string {
	const safePrefix = prefix
		.split("/")
		.filter(Boolean)
		.map((part) => part.replace(/[^a-zA-Z0-9_-]/g, "_"))
		.join("/");

	return [
		safePrefix || "uploads",
		randomUUIDv7(),
		sanitizeFilename(filename),
	].join("/");
}

import { createObjectKey, sanitizeFilename } from "./keys";
import type { createStorage } from "./uploads";

type PendingStorageObject = {
	id: string;
	objectKey: string;
	originalName: string;
	expectedSize: number;
	contentType: string;
	status: string;
};

type StorageRepository = {
	createPendingStorageObject(input: {
		objectKey: string;
		originalName: string;
		contentType: string;
		expectedSize: number;
	}): Promise<{ id: string; objectKey: string }>;

	findStorageObjectById(id: string): Promise<PendingStorageObject | null>;

	confirmStorageUpload(input: {
		id: string;
		verifiedSize: number;
		verifiedContentType: string;
	}): Promise<unknown>;

	markStorageUploadFailed(input: { id: string; reason: string }): Promise<void>;
};

type StorageClient = Pick<
	ReturnType<typeof createStorage>,
	| "createPreassignedUpload"
	| "uploadBuffer"
	| "headObject"
	| "getDownloadUrl"
	| "deleteObject"
	| "getPublicUrl"
>;

export type StorageErrorCode =
	| "NOT_FOUND"
	| "INVALID_STATE"
	| "OBJECT_MISSING"
	| "SIZE_MISMATCH"
	| "TYPE_MISMATCH";

export class StorageServiceError extends Error {
	readonly code: StorageErrorCode;

	constructor(code: StorageErrorCode, message: string) {
		super(message);
		this.name = "StorageServiceError";
		this.code = code;
	}
}

function isNotFound(error: unknown) {
	if (typeof error !== "object" || error === null) return false;

	const e = error as { name?: string; $metadata?: { httpStatusCode?: number } };

	return (
		e.name === "NotFound" ||
		e.name === "NoSuchKey" ||
		e.$metadata?.httpStatusCode === 404
	);
}

export function createStorageService(input: {
	storage: StorageClient;
	repository: StorageRepository;
}) {
	const { storage, repository } = input;

	async function markFailed({ id, reason }: { id: string; reason: string }) {
		await repository
			.markStorageUploadFailed({ id: id, reason: reason })
			.catch(() => {});
	}

	async function getRecord(id: string) {
		const record = await repository.findStorageObjectById(id);

		if (!record) {
			throw new StorageServiceError("NOT_FOUND", "Storage object not found");
		}

		return record;
	}

	async function uploadBuffer(file: {
		filename: string;
		contentType: string;
		body: Uint8Array;
		prefix?: string;
	}) {
		const objectKey = createObjectKey(file.filename, file.prefix);

		const pending = await repository.createPendingStorageObject({
			objectKey: objectKey,
			originalName: sanitizeFilename(file.filename),
			contentType: file.contentType,
			expectedSize: file.body.byteLength,
		});

		await storage.uploadBuffer({ ...file, objectKey });

		const uploaded = await storage.headObject(objectKey);

		const confirmed = await repository.confirmStorageUpload({
			id: pending.id,
			verifiedSize: uploaded.size,
			verifiedContentType: uploaded.contentType,
		});

		return {
			id: pending.id,
			objectKey: objectKey,
			uploaded: uploaded,
			confirmed: confirmed,
		};
	}

	async function getDownloadUrl(id: string, filename?: string) {
		const record = await getRecord(id);

		if (record.status !== "ACTIVE") {
			throw new StorageServiceError(
				"INVALID_STATE",
				`Object is not available (status ${record.status})`,
			);
		}

		return storage.getDownloadUrl(
			record.objectKey,
			filename ?? record.originalName,
		);
	}

	return {
		uploadBuffer,
    getPublicUrl: storage.getPublicUrl,
		getDownloadUrl
	};
}

import {
	DeleteObjectCommand,
	GetObjectCommand,
	HeadObjectCommand,
	PutObjectCommand,
} from "@aws-sdk/client-s3";
import { getSignedUrl } from "@aws-sdk/s3-request-presigner";

import { createS3Client } from "./client";
import { createObjectKey, sanitizeFilename } from "./keys";
import type { StorageConfig } from "./types";
import { validateFileBuffer, validateFileMetadata } from "./validation";

export function createStorage(config: StorageConfig) {
	const client = createS3Client(config);
	const maxSize = config.S3_MAX_FILE_SIZE_BYTES ?? 25 * 1024 * 1024;

	function validate(input: { contentType: string; size: number }) {
		validateFileMetadata({
			...input,
			maxSize,
		});
	}

	async function createPreassignedUpload(input: {
		filename: string;
		contentType: string;
		size: number;
		prefix?: string;
	}) {
		validate(input);

		const objectKey = createObjectKey(input.filename, input.prefix);

		const expiresIn = config.S3_UPLOAD_URL_EXPIRES_IN ?? 300;

		const uploadUrl = await getSignedUrl(
			client,
			new PutObjectCommand({
				Bucket: config.S3_BUCKET,
				Key: objectKey,
				ContentType: input.contentType,
			}),
			{ expiresIn },
		);

		return {
			objectKey,
			uploadUrl,
			expiresIn,
			originalName: sanitizeFilename(input.filename),
			contentType: input.contentType,
			size: input.size,
		};
	}

	async function uploadBuffer(input: {
		filename: string;
		contentType: string;
		body: Uint8Array;
		objectKey?: string;
		prefix?: string;
	}) {
		validate({
			contentType: input.contentType,
			size: input.body.byteLength,
		});

		await validateFileBuffer(input.body, input.contentType);

		const objectKey =
			input.objectKey ?? createObjectKey(input.filename, input.prefix);

		await client.send(
			new PutObjectCommand({
				Bucket: config.S3_BUCKET,
				Key: objectKey,
				Body: input.body,
				ContentType: input.contentType,
			}),
		);

		return {
			objectKey,
			originalName: sanitizeFilename(input.filename),
			contentType: input.contentType,
			size: input.body.byteLength,
		};
	}

	async function headObject(objectKey: string) {
		const result = await client.send(
			new HeadObjectCommand({
				Bucket: config.S3_BUCKET,
				Key: objectKey,
			}),
		);

		return {
			objectKey,
			contentType: result.ContentType ?? "application/octet-stream",
			size: result.ContentLength ?? 0,
			etag: result.ETag,
		};
	}

	async function getDownloadUrl(objectKey: string, filename?: string) {
		return getSignedUrl(
			client,
			new GetObjectCommand({
				Bucket: config.S3_BUCKET,
				Key: objectKey,
				ResponseContentDisposition: filename
					? `attachment; filename="${sanitizeFilename(filename)}"`
					: undefined,
			}),
			{ expiresIn: config.S3_DOWNLOAD_URL_EXPIRE_IN ?? 300 },
		);
	}

	function getPublicUrl(objectKey: string) {
		console.log(objectKey);
		if (!config.S3_PUBLIC_BASE_URL) {
			throw new Error("Public URLs are not configured");
		}

		const base = config.S3_PUBLIC_BASE_URL.endsWith("/")
			? config.S3_PUBLIC_BASE_URL
			: `${config.S3_PUBLIC_BASE_URL}/`;

		const encodedKey = objectKey.split("/").map(encodeURIComponent).join("/");

		return new URL(encodedKey, base).toString();
	}

	async function deleteObject(objectKey: string) {
		await client.send(
			new DeleteObjectCommand({
				Bucket: config.S3_BUCKET,
				Key: objectKey,
			}),
		);
	}

	return {
		createPreassignedUpload,
		uploadBuffer,
		headObject,
		getDownloadUrl,
		getPublicUrl,
		deleteObject,
	};
}

import { fileTypeFromBuffer } from "file-type";

const ALLOWED_TYPES = new Set([
	"image/jpeg",
	"image/png",
	"image/webp",
	"image/gif",
	"image/avif",
	"application/pdf",
	"text/plain",
	"text/csv",
]);

export function validateFileMetadata(input: {
	contentType: string;
	size: number;
	maxSize: number;
}) {
	if (!ALLOWED_TYPES.has(input.contentType)) {
		throw new Error("File type not allowed");
	}

	if (
		!Number.isSafeInteger(input.size) ||
		input.size <= 0 ||
		input.size > input.maxSize
	) {
		throw new Error("Invalid file size");
	}
}

export async function validateFileBuffer(
	buffer: Uint8Array,
	declaredType: string,
) {
	const detected = await fileTypeFromBuffer(buffer);

	if (detected && detected.mime !== declaredType) {
		throw new Error("File content does not match declared MIME type");
	}

	if (!detected && !declaredType.startsWith("text/")) {
		throw new Error("Unable to verify file signature");
	}

	if (declaredType.startsWith("text/")) {
		const text = new TextDecoder("utf-8", { fatal: true }).decode(buffer);

		if (text.includes("\0")) {
			throw new Error("Invalid text file");
		}
	}
}

// packages/storage/src/cleanup.ts

export type CleanupCandidate = {
	id: string;
	objectKey: string;
};

export type StorageCleanupRepository = {
	claimOrphans(input: {
		olderThan: Date;
		limit: number;
	}): Promise<CleanupCandidate[]>;

	markDeleted(id: string): Promise<void>;

	markDeleteFailed(id: string, error: string): Promise<void>;
};

export async function cleanupOrphanedObjects(input: {
	storage: {
		deleteObject(key: string): Promise<void>;
	};
	repository: StorageCleanupRepository;
	olderThan: Date;
	limit?: number;
}) {
	const candidates = await input.repository.claimOrphans({
		olderThan: input.olderThan,
		limit: input.limit ?? 100,
	});

	const results = {
		deleted: 0,
		failed: 0,
	};

	for (const candidate of candidates) {
		try {
			await input.storage.deleteObject(candidate.objectKey);
			await input.repository.markDeleted(candidate.id);
			results.deleted++;
		} catch (error) {
			results.failed++;

			await input.repository.markDeleteFailed(
				candidate.id,
				error instanceof Error ? error.message : "Unknown storage error",
			);
		}
	}

	return results;
}

		const storage = createStorage(config);

		const result = await storage.createPresignedUpload({
			filename: "foto.png",
			contentType: "image/png",
			size: 100,
			prefix: "users/123",
		});

		expect(result.objectKey).toMatch(/^users\/123\/[0-9a-f-]+\/foto\.png$/i);
	});

	test("rejeita MIME type não permitido", async () => {
		const storage = createStorage(config);

		await expect(
			storage.createPresignedUpload({
				filename: "script.exe",
				contentType: "application/x-executable",
				size: 100,
			}),
		).rejects.toThrow("File type not allowed");
	});

	test("rejeita arquivo maior que o limite configurado", async () => {
		const storage = createStorage(config);

		await expect(
			storage.createPresignedUpload({
				filename: "grande.pdf",
				contentType: "application/pdf",
				size: 1025,
			}),
		).rejects.toThrow("Invalid file size");
	});

	test("rejeita tamanho zero", async () => {
		const storage = createStorage(config);

		await expect(
			storage.createPresignedUpload({
				filename: "vazio.pdf",
				contentType: "application/pdf",
				size: 0,
			}),
		).rejects.toThrow("Invalid file size");
	});

	test("gera URL pública codificando cada segmento da chave", () => {
		const storage = createStorage({
			...config,
			publicBaseUrl: "https://cdn.example.com/files",
		});

		expect(storage.getPublicUrl("uploads/user 1/arquivo.pdf")).toBe(
			"https://cdn.example.com/files/uploads/user%201/arquivo.pdf",
		);
	});

	test("aceita base URL terminada em barra", () => {
		const storage = createStorage({
			...config,
			publicBaseUrl: "https://cdn.example.com/files/",
		});

		expect(storage.getPublicUrl("uploads/arquivo.pdf")).toBe(
			"https://cdn.example.com/files/uploads/arquivo.pdf",
		);
	});

	test("falha se a URL pública não estiver configurada", () => {
		const storage = createStorage(config);

		expect(() => storage.getPublicUrl("uploads/arquivo.pdf")).toThrow(
			"Public URLs are not configured",
		);
	});

	test("rejeita upload binário com assinatura inválida", async () => {
		const storage = createStorage(config);

		await expect(
			storage.uploadBuffer({
				filename: "arquivo.pdf",
				contentType: "application/pdf",
				body: new Uint8Array([1, 2, 3, 4]),
			}),
		).rejects.toThrow("Unable to verify file signature");
	});

	test("rejeita upload cujo conteúdo não corresponde ao MIME", async () => {
		const storage = createStorage(config);

		const png = new Uint8Array([
			0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a, 0x00, 0x00, 0x00, 0x0d,
			0x49, 0x48, 0x44, 0x52,
		]);

		await expect(
			storage.uploadBuffer({
				filename: "arquivo.jpg",
				contentType: "image/jpeg",
				body: png,
			}),
		).rejects.toThrow("File content does not match declared MIME type");
	});
});
